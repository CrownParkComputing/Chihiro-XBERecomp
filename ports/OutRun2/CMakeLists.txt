cmake_minimum_required(VERSION 3.25)
project(OutRun2Native LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include("${CMAKE_CURRENT_SOURCE_DIR}/cmake/Dependencies.cmake")
include("${CMAKE_CURRENT_SOURCE_DIR}/sdk/cmake/XbeRecomp.cmake")
file(GLOB XBE_HOST_SOURCES CONFIGURE_DEPENDS src/common/*.cpp)
add_xbe_image(OutRun2 outrun2.xbe game.toml 32 game)
target_compile_definitions(OutRun2 PRIVATE OUTRUN2_SDL_INPUT=1)
set_target_properties(OutRun2 PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")

enable_testing()
find_package(Python3 3.11 REQUIRED COMPONENTS Interpreter)
add_test(NAME ChihiroImportTests COMMAND "${Python3_EXECUTABLE}" -m unittest discover -s tests -v)
set_tests_properties(ChihiroImportTests PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}")
